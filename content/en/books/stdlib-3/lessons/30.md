## Assembling the CLI with argparse Subcommands

From Chapter 4, recall `argparse`. Our tool needs three subcommands: `analyze` (analyse and display), `export` (write to a file), and `stats` (print a summary).

```python
import argparse
import json
import logging
import sys
from collections import Counter, defaultdict

from dataclasses import dataclass


@dataclass
class LogEntry:
    timestamp: str
    level: str
    module: str
    message: str

    @property
    def hour(self):
        return self.timestamp.split(" ")[1].split(":")[0]


LOG_PATTERN = re_compiled_pattern = None  # placeholder


def parse_line(line: str) -> LogEntry | None:
    line = line.strip()
    if not line:
        return None
    match = re.match(
        r'^(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}) '
        r'\[(\w+)\] '
        r'\[(\w+)\] '
        r'(.*)$',
        line,
    )
    if match is None:
        return None
    return LogEntry(match.group(1), match.group(2), match.group(3), match.group(4).strip())


def parse_lines(lines):
    entries, failed = [], []
    for line in lines:
        e = parse_line(line)
        if e is None:
            failed.append(line.strip())
        else:
            entries.append(e)
    return entries, failed


def build_stats(entries):
    return {
        "total": len(entries),
        "level_counts": dict(Counter(e.level for e in entries)),
        "hourly_counts": dict(sorted(Counter(e.hour for e in entries).items())),
        "top_errors": Counter(
            e.message for e in entries if e.level == "ERROR"
        ).most_common(5),
        "module_errors": dict(sorted(
            ((m, c) for m, c in defaultdict(int, {}).items()) if False else
            sorted(defaultdict(int, {}).items(), key=lambda x: x[1], reverse=True)
        )),
    }


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="logalyzer",
        description="A command-line log analysis tool",
        formatter_class=argparse.RawDescriptionHelpFormatter,
    )
    parser.add_argument("--version", action="version", version="logalyzer 0.1.0")
    parser.add_argument("--verbose", "-v", action="store_true",
                        help="show detailed runtime logs")

    sub = parser.add_subparsers(dest="command")

    # analyze subcommand
    p_analyze = sub.add_parser("analyze", help="analyse logs and print a console report")
    p_analyze.add_argument("input", help="log file path (leave empty to read stdin)")
    p_analyze.add_argument("--top", type=int, default=5, help="number of top errors")

    # export subcommand
    p_export = sub.add_parser("export", help="export statistics to a file")
    p_export.add_argument("input", help="log file path")
    p_export.add_argument("-f", "--format", choices=["json", "csv"],
                          default="json", help="export format")
    p_export.add_argument("-o", "--output", help="output file path")

    # stats subcommand
    p_stats = sub.add_parser("stats", help="print statistics summary as JSON only")
    p_stats.add_argument("input", help="log file path")

    return parser
```

## The Main Flow

```python
def run_analyze(args, logger) -> int:
    """Implementation of the analyze subcommand"""
    with open(args.input, encoding="utf-8") as f:
        content = f.read()
    lines = content.splitlines()
    logger.info(f"Read {len(lines)} log lines")

    entries, failed = parse_lines(lines)
    logger.info(f"Parsing complete: {len(entries)} succeeded, {len(failed)} failed")
    if failed:
        logger.warning(f"{len(failed)} lines could not be parsed")

    stats = build_stats(entries)

    # console output
    print("=" * 50)
    print("Log analysis report")
    print("=" * 50)
    print(f"\nTotal: {stats['total']} log entries\n")

    print("[Level distribution]")
    total = stats["total"]
    for level, count in sorted(stats["level_counts"].items(),
                                key=lambda x: x[1], reverse=True):
        ratio = count / total * 100 if total else 0
        print(f"  {level:<8} {count:<6} {ratio:.1f}%")

    print(f"\n[Top {args.top} errors]")
    for i, (msg, count) in enumerate(stats["top_errors"][:args.top], 1):
        print(f"  {i}. [{count} times] {msg}")

    return 0


def run_export(args, logger) -> int:
    """Implementation of the export subcommand"""
    with open(args.input, encoding="utf-8") as f:
        entries, failed = parse_lines(f.read().splitlines())

    stats = build_stats(entries)
    logger.info(f"Parsed {len(entries)} entries, {len(failed)} failed")

    if args.format == "json":
        output = json.dumps(stats, ensure_ascii=False, indent=2)
        ext = "json"
    else:
        import csv, io
        buf = io.StringIO()
        writer = csv.writer(buf)
        writer.writerow(["type", "name", "count"])
        for level, count in stats["level_counts"].items():
            writer.writerow(["level", level, count])
        output = buf.getvalue()
        ext = "csv"

    if args.output:
        with open(args.output, "w", encoding="utf-8") as f:
            f.write(output)
        logger.info(f"Written to {args.output}")
    else:
        out_path = f"report.{ext}"
        with open(out_path, "w", encoding="utf-8") as f:
            f.write(output)
        logger.info(f"Written to {out_path}")

    return 0


def run_stats(args, logger) -> int:
    """stats subcommand: JSON summary only"""
    with open(args.input, encoding="utf-8") as f:
        entries, _ = parse_lines(f.read().splitlines())

    stats = build_stats(entries)
    print(json.dumps({
        "total": stats["total"],
        "level_counts": stats["level_counts"],
        "top_errors": stats["top_errors"],
    }, ensure_ascii=False, indent=2))
    return 0


def main(argv=None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)

    logger = logging.getLogger("logalyzer")
    level = logging.DEBUG if getattr(args, "verbose", False) else logging.INFO
    logging.basicConfig(level=level, format="[%(levelname)s] %(message)s")

    if args.command is None:
        parser.print_help()
        return 0

    handlers = {
        "analyze": run_analyze,
        "export": run_export,
        "stats": run_stats,
    }
    handler = handlers.get(args.command)
    if handler is None:
        parser.error(f"Unknown subcommand: {args.command}")
        return 1

    return handler(args, logger)


if __name__ == "__main__":
    sys.exit(main())
```

## Usage

```bash
# show help
$ python -m logalyzer --help

# analyse logs
$ python -m logalyzer analyze app.log --top 10

# export as JSON
$ python -m logalyzer export app.log -f json -o report.json

# export as CSV
$ python -m logalyzer export app.log -f csv -o report.csv

# print statistics summary only
$ python -m logalyzer stats app.log

# verbose mode
$ python -m logalyzer -v analyze app.log
```

## Testing With unittest

From Chapter 5, unittest now tests our tool:

```python
import unittest
from logalyzer.parser import parse_line, parse_lines
from logalyzer.stats import (
    level_distribution, hourly_distribution,
    top_errors, module_error_count,
)


class TestParser(unittest.TestCase):
    def test_parse_valid_line(self):
        line = "2026-01-15 10:23:45 [ERROR] [auth] login failed"
        entry = parse_line(line)
        self.assertIsNotNone(entry)
        self.assertEqual(entry.level, "ERROR")
        self.assertEqual(entry.module, "auth")
        self.assertEqual(entry.message, "login failed")

    def test_parse_invalid_line(self):
        self.assertIsNone(parse_line("this line has the wrong format"))
        self.assertIsNone(parse_line(""))
        self.assertIsNone(parse_line("   "))

    def test_parse_lines_partial(self):
        lines = [
            "2026-01-15 10:23:45 [INFO] [order] ok",
            "garbage line",
            "2026-01-15 10:23:46 [ERROR] [auth] fail",
            "",
        ]
        entries, failed = parse_lines(lines)
        self.assertEqual(len(entries), 2)
        self.assertEqual(len(failed), 2)
        self.assertEqual(entries[0].level, "INFO")
        self.assertEqual(entries[1].level, "ERROR")


class TestStats(unittest.TestCase):
```

## Crash Site: A Subcommand Argument Parse Error — "too few arguments"

```python
# WRONG: forgetting required=True or dest
sub = parser.add_subparsers()
p_analyze = sub.add_parser("analyze")
p_analyze.add_argument("input")  # without required, the default is already required
# but the user runs `logalyzer analyze` with no file path
```

**Error output**:

```
usage: logalyzer analyze [-h] input
logalyzer analyze: error: the following arguments are required: input
```

**This is the expected behaviour** — but if you want `input` to be optional (read from stdin instead), set `nargs="?"`:

```python
p_analyze.add_argument("input", nargs="?", default=None,
                       help="log file path (leave empty to read stdin)")
```

## Crash Site: `main()` Opens Files Directly, Making Unit Tests Impossible to Cover

```python
# WRONG: reading the file directly inside the main flow
def run_analyze(args):
    with open(args.input) as f:       # tests now need a real file on disk
        entries, _ = parse_lines(f.read().splitlines())
    ...
```

**The problem**: covering `run_analyze` in a unit test means creating temporary files, which makes tests slow and brittle.

**The fix**: separate "reading the file" from "processing the logic," so the logic accepts a list of strings:

```python
def analyze_logic(lines: list[str]) -> dict:
    """Pure logic, no file access, directly unit-testable"""
    entries, failed = parse_lines(lines)
    return build_stats(entries)

def run_analyze(args):
    with open(args.input) as f:
        return analyze_logic(f.read().splitlines())  # a thin I/O wrapper
```

- Files and paths: `pathlib` / `os` / `shutil` / `csv` / `json`

- Networking basics: `socket` / `urllib` / `http.server`

- Regex engine: compiling patterns, groups, greediness, and backtracking in the `re` module

**Together, how much ground have we covered?** The Python standard library has 200+ modules; we've gone deep on about 40 core ones. **But those 40 cover roughly 80% of day-to-day development.** For the remaining modules, you now know how to read the documentation and get started.

## Crash Site: Stalling After Finishing the Standard Library

```python

```

**Reality**: the `requests` library does the same job in about 10 lines. The standard library is the foundation, but **you build the house on that foundation with more productive tools.** Knowing what the standard library can do — and when a third-party library is the better choice — is what makes a mature engineer.

**The right mindset**: standard library = understanding the fundamentals + a fallback option; third-party libraries = everyday productivity. Both matter.

## Summary

- argparse subcommands wire the parsing, statistics, and output layers into a complete CLI tool.
- Subcommands are registered with `add_subparsers()` + `add_parser()`, each with its own arguments.
- Keep the main flow function "thin": it handles only I/O; the logic lives in pure functions (easy to test).
- unittest can test the parsing and statistics layers, with each layer covered independently.
- The three volumes cover 40 core modules, enough for 80% of real-world scenarios.
- Next steps by direction: data analysis / algorithm competitions / desktop applications / web engineering.
