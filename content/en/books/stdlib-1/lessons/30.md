## Assembling the Complete Pipeline

The four layers are in place — now to wire them together:

```python
import sys
import json
import csv
import configparser
from decimal import Decimal, InvalidOperation
from enum import Enum
import logging

# ===== configure logging =====
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    datefmt="%H:%M:%S",
)

# ===== domain model =====
class Status(Enum):
    PENDING = "pending"
    PAID = "paid"
    SHIPPED = "shipped"

# ===== load layer =====
def load_json(filepath):
    with open(filepath, encoding="utf-8") as f:
        data = json.load(f)
    return data if isinstance(data, list) else [data]

def load_csv(filepath):
    with open(filepath, encoding="utf-8", newline="") as f:
        return list(csv.DictReader(f))

def load_ini(filepath):
    cp = configparser.ConfigParser()
    cp.read(filepath, encoding="utf-8")
    records = []
    for section in cp.sections():
        d = dict(cp[section])
        d["id"] = section
        records.append(d)
    return records

def load(filepath):
    if filepath.endswith(".json"):
        return load_json(filepath)
    elif filepath.endswith(".csv"):
        return load_csv(filepath)
    elif filepath.endswith(".ini"):
        return load_ini(filepath)
    else:
        raise ValueError(f"unsupported format: {filepath}")

# ===== transform layer =====
def normalize(raw_list):
    result = []
    for raw in raw_list:
        id_val = raw.get("id") or raw.get("order_no") or raw.get("no") or ""
        amount = raw.get("amount") or raw.get("amount_gbp") or "0"
        status = raw.get("status") or raw.get("state") or "pending"
        qty = raw.get("qty") or raw.get("quantity") or "0"
        result.append({
            "id": str(id_val),
            "amount": str(amount),
            "status": str(status),
            "qty": int(qty) if str(qty).strip() else 0,
        })
    return result

# ===== check layer =====
def validate(record):
    errors = []
    if not record.get("id"):
        errors.append("missing id")
    try:
        amount = Decimal(record["amount"])
    except InvalidOperation:
        errors.append(f"invalid amount: {record.get('amount')}")
        amount = None
    if amount is not None and amount < 0:
        errors.append(f"amount cannot be negative: {amount}")
    qty = record.get("qty", 0)
    if qty <= 0:
        errors.append(f"quantity must be greater than 0: {qty}")
    try:
        Status(record.get("status"))
    except ValueError:
        errors.append(f"invalid status: {record.get('status')}")
    return errors

def check_all(records):
    passed = []
    failed = []
    for rec in records:
        errors = validate(rec)
        if errors:
            failed.append({"record": rec, "errors": errors})
        else:
            passed.append(rec)
    return passed, failed

# ===== dump layer =====
def record_to_dict(rec):
    return {
        "id": rec["id"],
        "amount": str(rec["amount"]),
        "status": rec["status"],
        "qty": int(rec["qty"]),
    }

def dump_json(records, filepath):
    data = [record_to_dict(r) for r in records]
    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def dump_csv(records, filepath):
    fieldnames = ["id", "amount", "status", "qty"]
    with open(filepath, "w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for rec in records:
            writer.writerow(record_to_dict(rec))

def dump(records, filepath):
    if filepath.endswith(".json"):
        dump_json(records, filepath)
    elif filepath.endswith(".csv"):
        dump_csv(records, filepath)
    else:
        raise ValueError(f"unsupported output format: {filepath}")

# ===== report =====
def make_report(passed, failed):
    lines = [
        "=== Data Validation Report ===",
        f"total: {len(passed) + len(failed)} records",
        f"passed: {len(passed)} records",
        f"failed: {len(failed)} records",
        "",
    ]
    if failed:
        lines.append("--- failure details ---")
        for f in failed:
            rid = f["record"].get("id") or "(no id)"
            lines.append(f"[{rid}]")
            for err in f["errors"]:
                lines.append(f"  ✗ {err}")
    return "\n".join(lines)

# ===== main flow =====
def run(input_path, output_path):
    logging.info(f"reading input: {input_path}")
    raw = load(input_path)
    logging.info(f"read {len(raw)} raw records")

    records = normalize(raw)
    logging.info(f"normalisation complete: {len(records)} records")

    passed, failed = check_all(records)
    logging.info(f"validation complete: {len(passed)} passed, {len(failed)} failed")

    if failed:
        logging.warning(f"{len(failed)} records failed validation — see report")

    if passed:
        logging.info(f"writing output: {output_path}")
        dump(passed, output_path)

    return make_report(passed, failed)

# ===== CLI entry point =====
def main():
    if len(sys.argv) < 3:
        print("usage: python pipe.py <input_file> <output_file>")
        print("example: python pipe.py data.json output.csv")
        sys.exit(1)

    input_path = sys.argv[1]
    output_path = sys.argv[2]

    try:
        report = run(input_path, output_path)
        print(report)
        print("\n✓ processing complete")
    except FileNotFoundError:
        logging.error(f"file not found: {input_path}")
        sys.exit(1)
    except ValueError as e:
        logging.error(str(e))
        sys.exit(1)

if __name__ == "__main__":
    main()
```

**This is the complete pipeline tool.** Each layer has a single responsibility, and the main flow simply connects them. `sys.argv` handles the command-line arguments.

## How sys.argv Works

```python
import sys

# command: python pipe.py data.json output.csv
print(sys.argv)
# ['pipe.py', 'data.json', 'output.csv']

# sys.argv[0] is the script name; sys.argv[1:] are the arguments
script_name = sys.argv[0]
args = sys.argv[1:]
```

**`sys.argv` is a list of strings.** `sys.argv[0]` is always the script path; the real arguments start at `[1]`.

Why not `argparse`? Because `sys.argv` is enough — two arguments, no subcommands, no defaults, no `--verbose` flags needed. `argparse` belongs in the standard library volume 3, which you'll meet there.

### JSON: The Universal Language of Data Exchange

`json.dump(data, f, ensure_ascii=False, indent=2)` — Chinese characters unescaped, with indentation. `Decimal` and `Enum` can't be serialised directly; you need a custom `default` function or manual type conversion.

### CSV: A Table Is a List

`csv.DictReader` turns each row into a dict; `csv.DictWriter` writes dicts as rows. Always open with `newline=""`.

### configparser: Reading and Writing Configuration Files

The INI format uses sections; every value is a string and must be manually converted.

### pickle: Binary Serialisation of Python Objects

It can serialise nearly any Python object, but it isn't safe — **never unpickle untrusted data.**

### base64: Binary to Text

Encodes bytes into an ASCII string for transmitting binary data across networks or text protocols.

### hashlib: Data Integrity Verification

`hashlib.sha256(b"data").hexdigest()` — detects whether a file has been tampered with.

### Decimal: The Right Tool for Money

`Decimal("0.1") + Decimal("0.2")` equals `0.3` exactly, whereas `0.1 + 0.2` gives `0.30000000000000004`.

### Enum: Killing Magic Strings

`Status.PAID.value` gets the value; `Status("paid")` looks it up. Illegal values raise `ValueError`.

### dataclasses: Less Boilerplate

`@dataclass` auto-generates `__init__`, `__repr__`, and friends. `asdict()` does shallow serialisation.

## Crash Site 1: sys.argv Index Out of Range

```python
import sys

# WRONG: accessing an index without checking first
# input_path = sys.argv[1]   # raises IndexError if no argument was passed

# correct: check the length first
if len(sys.argv) < 2:
    print("usage: python script.py <file>")
    sys.exit(1)
input_path = sys.argv[1]
```

**`sys.argv` raises `IndexError` without bounds checking.** The first step in any CLI tool is always to validate the argument count.

## Crash Site 2: Forgetting `if __name__ == "__main__"`

```python
# WRONG: calling main() directly
# main()  # if this file is imported, main() runs too

# correct
if __name__ == "__main__":
    main()
```

**`if __name__ == "__main__":` guarantees that `main()` only runs when the script is executed directly.** If the module is imported elsewhere, the main flow won't be triggered. It's basic etiquette for Python modules.
