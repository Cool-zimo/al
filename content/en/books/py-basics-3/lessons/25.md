# Practice: a student record system

> Combine classes with file persistence to build something genuinely structured.

## What we're building

A **student record system** that can:
- add a student (name, id, scores)
- list everyone
- search by name
- show the class average
- save to and load from JSON

**The point**: organise it with classes rather than a pile of loose functions.

## Designing the classes

Decide what things exist:

| Class | Owns | Attributes | Methods |
|---|---|---|---|
| `Student` | one student | name, sid, scores | add_score, average |
| `Classroom` | one class | name, students | add, find, average, save, load |

**Two classes, one layer each** — that's layered design.

## The Student class

```python
class Student:
    """One student"""

    def __init__(self, name, sid):
        self.name = name
        self.sid = sid
        self.scores = []

    def add_score(self, score):
        if not 0 <= score <= 100:
            raise ValueError(f"score must be 0-100, got {score}")
        self.scores.append(score)

    @property
    def average(self):
        if not self.scores:
            return None
        return sum(self.scores) / len(self.scores)

    def to_dict(self):
        """Convert to a dict so it can be stored as JSON."""
        return {'name': self.name, 'sid': self.sid, 'scores': self.scores}

    @classmethod
    def from_dict(cls, d):
        """Rebuild an object from a dict."""
        s = cls(d['name'], d['sid'])
        s.scores = d['scores']
        return s

    def __str__(self):
        avg = self.average
        text = f"{avg:.1f}" if avg is not None else "no scores"
        return f"{self.name} ({self.sid}) average {text}"

    def __repr__(self):
        return f"Student(name='{self.name}', sid='{self.sid}')"
```

**Points to note:**

- **`average` is a `@property`** — it's derived data, not an action
- **`to_dict` / `from_dict`** — the standard way to move between objects and JSON
- **`@classmethod`** — `cls` is the class itself, used to build an object of this class

## The Classroom class

```python
import json


class Classroom:
    """One class of students"""

    def __init__(self, name):
        self.name = name
        self.students = []

    def add(self, student):
        if self.find(student.sid):
            raise ValueError(f"id {student.sid} already exists")
        self.students.append(student)

    def find(self, sid):
        for s in self.students:
            if s.sid == sid:
                return s
        return None

    def search(self, keyword):
        return [s for s in self.students if keyword in s.name]

    @property
    def average(self):
        avgs = [s.average for s in self.students if s.average is not None]
        if not avgs:
            return None
        return sum(avgs) / len(avgs)

    def save(self, filename):
        data = [s.to_dict() for s in self.students]
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def load(self, filename):
        try:
            with open(filename, 'r', encoding='utf-8') as f:
                content = f.read()
        except FileNotFoundError:
            return          # first run — no file is normal

        if not content.strip():
            return

        try:
            data = json.loads(content)
        except json.JSONDecodeError:
            print("corrupt file — starting empty")
            return

        self.students = [Student.from_dict(d) for d in data]

    def __str__(self):
        return f"{self.name}: {len(self.students)} students"
```

## The main program

```python
FILE = 'classroom.json'


def add_student(room):
    name = input("name: ").strip()
    if not name:
        print("name can't be empty")
        return
    sid = input("id: ").strip()

    student = Student(name, sid)
    try:
        room.add(student)
    except ValueError as e:
        print(f"  ✗ {e}")
        return

    print("enter scores (blank line to finish):")
    while True:
        raw = input("  > ").strip()
        if not raw:
            break
        try:
            student.add_score(float(raw))
        except ValueError as e:
            print(f"    {e}")

    print(f"  ✓ added {student}")


def show_all(room):
    if not room.students:
        print("no students yet")
        return
    for i, s in enumerate(room.students, 1):
        print(f"{i}. {s}")


def main():
    room = Classroom("Class 1")
    room.load(FILE)
    print(room)

    while True:
        print("\n1.add  2.list  3.search  4.class average  5.save  6.quit")
        choice = input("choose: ").strip()

        if choice == '1':
            add_student(room)
        elif choice == '2':
            show_all(room)
        elif choice == '3':
            key = input("keyword: ").strip()
            for s in room.search(key):
                print(" ", s)
        elif choice == '4':
            avg = room.average
            print(f"class average: {avg:.1f}" if avg is not None else "no scores yet")
        elif choice == '5':
            room.save(FILE)
            print("  ✓ saved")
        elif choice == '6':
            room.save(FILE)
            print("bye")
            break
        else:
            print("please enter 1-6")


if __name__ == "__main__":
    main()
```

## Why classes make this better

**Compare the loose-function version:**

```python
# without classes
students = [{'name': 'Alice', 'sid': '001', 'scores': []}]
def add_score(students, sid, score): ...
def find(students, sid): ...
def average(students): ...
```

**Problems:**
- every function takes `students`
- the shape of `students` is convention only
- adding a "class name" field means editing everywhere

**With classes:**
- `room.add(s)` — the data lives on the object
- the shape is written in the class, obvious at a glance
- new fields touch only the class

## One key design: to_dict / from_dict

**JSON stores only basic types — it can't store your objects.**

```python
# ❌ doesn't work
json.dump(room.students, f)      # TypeError: Object of type Student is not JSON serializable
```

**So you convert both ways:**

```python
# save: object → dict → JSON
data = [s.to_dict() for s in students]
json.dump(data, f)

# load: JSON → dict → object
data = json.load(f)
students = [Student.from_dict(d) for d in data]
```

**This is the standard pattern for persisting objects** — database ORMs work the same way.
