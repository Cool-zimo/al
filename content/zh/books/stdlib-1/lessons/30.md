## 拼装完整管道

四层已经各就各位，现在把它们串起来：

```python
import sys
import json
import csv
import configparser
from decimal import Decimal, InvalidOperation
from enum import Enum
import logging

# ===== 配置日志 =====
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    datefmt="%H:%M:%S",
)

# ===== 领域模型 =====
class Status(Enum):
    PENDING = "pending"
    PAID = "paid"
    SHIPPED = "shipped"

# ===== 读取层 =====
def load_json(filepath):
    with open(filepath, encoding="utf-8") as f:
        data = json.load(f)
    return data if isinstance(data, list) else [data]

def load_csv(filepath):
    with open(filepath, encoding="utf-8", newline="") as f:
        return list(csv.DictReader(f))

def load_ini(filepath):
    cp = configparser.ConfigParser()
    cp.read(filepath, encoding="utf-8")
    records = []
    for section in cp.sections():
        d = dict(cp[section])
        d["id"] = section
        records.append(d)
    return records

def load(filepath):
    if filepath.endswith(".json"):
        return load_json(filepath)
    elif filepath.endswith(".csv"):
        return load_csv(filepath)
    elif filepath.endswith(".ini"):
        return load_ini(filepath)
    else:
        raise ValueError(f"不支持的格式: {filepath}")

# ===== 转换层 =====
def normalize(raw_list):
    result = []
    for raw in raw_list:
        id_val = raw.get("id") or raw.get("订单号") or raw.get("编号") or ""
        amount = raw.get("amount") or raw.get("金额") or "0"
        status = raw.get("status") or raw.get("状态") or "pending"
        qty = raw.get("qty") or raw.get("数量") or "0"
        result.append({
            "id": str(id_val),
            "amount": str(amount),
            "status": str(status),
            "qty": int(qty) if str(qty).strip() else 0,
        })
    return result

# ===== 校验层 =====
def validate(record):
    errors = []
    if not record.get("id"):
        errors.append("缺少 id")
    try:
        amount = Decimal(record["amount"])
    except InvalidOperation:
        errors.append(f"金额非法: {record.get('amount')}")
        amount = None
    if amount is not None and amount < 0:
        errors.append(f"金额不能为负: {amount}")
    qty = record.get("qty", 0)
    if qty <= 0:
        errors.append(f"数量必须大于 0: {qty}")
    try:
        Status(record.get("status"))
    except ValueError:
        errors.append(f"状态非法: {record.get('status')}")
    return errors

def check_all(records):
    passed = []
    failed = []
    for rec in records:
        errors = validate(rec)
        if errors:
            failed.append({"record": rec, "errors": errors})
        else:
            passed.append(rec)
    return passed, failed

# ===== 输出层 =====
def record_to_dict(rec):
    return {
        "id": rec["id"],
        "amount": str(rec["amount"]),
        "status": rec["status"],
        "qty": int(rec["qty"]),
    }

def dump_json(records, filepath):
    data = [record_to_dict(r) for r in records]
    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def dump_csv(records, filepath):
    fieldnames = ["id", "amount", "status", "qty"]
    with open(filepath, "w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for rec in records:
            writer.writerow(record_to_dict(rec))

def dump(records, filepath):
    if filepath.endswith(".json"):
        dump_json(records, filepath)
    elif filepath.endswith(".csv"):
        dump_csv(records, filepath)
    else:
        raise ValueError(f"不支持的输出格式: {filepath}")

# ===== 报告 =====
def make_report(passed, failed):
    lines = [
        "=== 数据校验报告 ===",
        f"总计: {len(passed) + len(failed)} 条",
        f"通过: {len(passed)} 条",
        f"失败: {len(failed)} 条",
        "",
    ]
    if failed:
        lines.append("--- 失败详情 ---")
        for f in failed:
            rid = f["record"].get("id") or "(无id)"
            lines.append(f"[{rid}]")
            for err in f["errors"]:
                lines.append(f"  ✗ {err}")
    return "\n".join(lines)

# ===== 主流程 =====
def run(input_path, output_path):
    logging.info(f"读取输入: {input_path}")
    raw = load(input_path)
    logging.info(f"读取到 {len(raw)} 条原始记录")

    records = normalize(raw)
    logging.info(f"标准化完成: {len(records)} 条")

    passed, failed = check_all(records)
    logging.info(f"校验完成: 通过 {len(passed)} 条, 失败 {len(failed)} 条")

    if failed:
        logging.warning(f"{len(failed)} 条记录校验失败，详见报告")

    if passed:
        logging.info(f"写入输出: {output_path}")
        dump(passed, output_path)

    return make_report(passed, failed)

# ===== 命令行入口 =====
def main():
    if len(sys.argv) < 3:
        print("用法: python pipe.py <输入文件> <输出文件>")
        print("示例: python pipe.py data.json output.csv")
        sys.exit(1)

    input_path = sys.argv[1]
    output_path = sys.argv[2]

    try:
        report = run(input_path, output_path)
        print(report)
        print("\n✓ 处理完成")
    except FileNotFoundError:
        logging.error(f"文件不存在: {input_path}")
        sys.exit(1)
    except ValueError as e:
        logging.error(str(e))
        sys.exit(1)

if __name__ == "__main__":
    main()
```

**这就是完整的管道工具**。四层各司其职，主流程只是把它们串起来。`sys.argv` 负责取命令行参数。

## sys.argv 怎么用

```python
import sys

# 命令行: python pipe.py data.json output.csv
print(sys.argv)
# ['pipe.py', 'data.json', 'output.csv']

# sys.argv[0] 是脚本名，sys.argv[1:] 是参数
script_name = sys.argv[0]
args = sys.argv[1:]
```

**`sys.argv` 是一个字符串列表**。`sys.argv[0]` 永远是脚本路径，真正的参数从 `[1]` 开始。

为什么不用 `argparse`？因为 `sys.argv` 够用了——两个参数，不需要子命令、不需要默认值、不需要 `--verbose` 这种选项。`argparse` 是标准库 3 的内容，届时你会学到。

## 全书回顾：你带走了什么

| 章节 | 核心技能 | 关键工具 |
|------|---------|---------|
| 第 1 章 | JSON 读写与序列化 | `json.dump/load`、`ensure_ascii` |
| 第 2 章 | JSON 进阶 | `default` 参数、自定义编码器 |
| 第 3 章 | 表格数据处理 | `csv.reader/writer/DictReader` |
| 第 4 章 | 配置与持久化 | `configparser`、`pickle`、`base64`、`hashlib` |
| 第 5 章 | 精确计算与类型安全 | `Decimal`、`Fraction`、`dataclasses`、`Enum` |
| 第 6 章 | 工程化思维 | 分层设计、日志、命令行工具 |

### JSON：数据交换的通用语言

`json.dump(data, f, ensure_ascii=False, indent=2)`——中文不转义、带缩进。`Decimal` 和 `Enum` 不能直接序列化，需要自定义 `default` 函数或手动转类型。

### CSV：表格即列表

`csv.DictReader` 把每行变成 dict，`csv.DictWriter` 把 dict 写成行。打开必须 `newline=""`。

### configparser：配置文件的读写

INI 格式分 section，所有值都是字符串，需要手动转类型。

### pickle：Python 对象的二进制序列化

能序列化几乎所有 Python 对象，但不安全——**不要 unpickle 不信任的数据**。

### base64：二进制转文本

把字节编码成 ASCII 字符串，用于在网络或文本协议中传输二进制。

### hashlib：数据完整性校验

`hashlib.sha256(b"data").hexdigest()`——检测文件是否被篡改。

### Decimal：算钱就用它

`Decimal("0.1") + Decimal("0.2")` 精确等于 `0.3`，而 `0.1 + 0.2` 是 `0.30000000000000004`。

### Enum：杜绝魔法字符串

`Status.PAID.value` 取值，`Status("paid")` 反查。非法值抛 `ValueError`。

### dataclasses：少写样板代码

`@dataclass` 自动生成 `__init__`、`__repr__` 等。`asdict()` 做浅层序列化。

## 翻车现场一：sys.argv 索引越界

```python
import sys

# WRONG：直接取索引不做检查
# input_path = sys.argv[1]   # 如果没传参数 → IndexError

# 正确：先判断长度
if len(sys.argv) < 2:
    print("用法: python script.py <文件>")
    sys.exit(1)
input_path = sys.argv[1]
```

**`sys.argv` 不做边界检查就会 `IndexError`**。CLI 工具的第一步永远是检查参数个数。

## 翻车现场二：忘记 if __name__ == "__main__"

```python
# WRONG：直接调用 main()
# main()  # 如果这个文件被 import，main() 也会执行

# 正确
if __name__ == "__main__":
    main()
```

**`if __name__ == "__main__":` 保证只有直接运行脚本时才执行 `main()`**。如果被其他模块 import，不会触发主流程。这是 Python 模块的基本礼仪。
