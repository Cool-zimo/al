# 输出层：多格式输出与工具自日志

> 这一课让统计结果"看得见、存得下"——控制台表格、JSON 导出、CSV 导出三管齐下，同时用 logging 记录工具自身的运行日志。

## 输出层要做什么

统计层产出的数据是 Python 字典，人看不懂，也不能发给别人。输出层负责把数据变成：
1. **控制台表格**——开发调试时直接看
2. **JSON**——方便程序间传递、Web API 返回
3. **CSV**——方便导入 Excel 做进一步分析

输出层的输入是纯字典/列表，输出是字符串（或写入文件）。它不知道数据从哪来，也不关心。

## 控制台表格

```python
def format_stats_table(stats: dict) -> str:
    """把统计结果格式化为控制台表格"""
    lines = []

    # 级别分布表
    lines.append("=" * 50)
    lines.append("📊 日志级别分布")
    lines.append("=" * 50)
    lines.append(f"{'级别':<10} {'数量':<10} {'占比':<10}")
    lines.append("-" * 50)
    total = sum(stats["level_counts"].values())
    for level, count in sorted(stats["level_counts"].items(),
                                key=lambda x: x[1], reverse=True):
        ratio = count / total * 100 if total else 0
        lines.append(f"{level:<10} {count:<10} {ratio:.1f}%")

    # 小时分布
    lines.append("")
    lines.append("=" * 50)
    lines.append("🕐 每小时日志量")
    lines.append("=" * 50)
    for hour, count in stats["hourly_counts"].items():
        bar = "█" * count
        lines.append(f"  {hour}:00 | {bar} {count}")

    # Top 错误
    lines.append("")
    lines.append("=" * 50)
    lines.append("🚨 Top 5 错误")
    lines.append("=" * 50)
    for i, (msg, count) in enumerate(stats["top_errors"], 1):
        lines.append(f"  {i}. [{count:>2}次] {msg}")

    return "\n".join(lines)


# 使用示例
stats = {
    "level_counts": {"INFO": 8, "ERROR": 8, "WARN": 3, "DEBUG": 2},
    "hourly_counts": {"10": 6, "11": 5, "12": 4, "13": 5},
    "top_errors": [("用户登录失败", 3), ("连接超时", 2), ("token 过期", 2)],
}
print(format_stats_table(stats))
```

输出：

```
--------------------------------------------------
```

## JSON 导出

```python
import json


def export_json(stats: dict, indent: int = 2) -> str:
    """导出为 JSON 字符串"""
    return json.dumps(stats, ensure_ascii=False, indent=indent)


json_str = export_json(stats)
print(json_str[:200])
```

输出：

```json
{
  "level_counts": {
    "INFO": 8,
    "ERROR": 8,
    "WARN": 3,
    "DEBUG": 2
  },
  "hourly_counts": {
    "10": 6,
    "11": 5,
    "12": 4,
    "13": 5
  },
```

**`ensure_ascii=False` 是关键**——不设的话中文会变成 `\u7528\u6237\u767b\u5f55\u5931\u8d25`，人根本看不懂。

## CSV 导出

```python
import csv
import io


def export_csv(stats: dict) -> str:
    """导出为 CSV 字符串"""
    buf = io.StringIO()
    writer = csv.writer(buf)

    # 写入级别分布
    writer.writerow(["类型", "名称", "数量"])
    for level, count in stats["level_counts"].items():
        writer.writerow(["级别", level, count])

    # 写入 Top 错误
    for msg, count in stats["top_errors"]:
        writer.writerow(["错误", msg, count])

    return buf.getvalue()


csv_str = export_csv(stats)
print(csv_str)
```

输出：

## 工具自身的日志：用 logging 记录运行过程

注意区分两个概念：
- **业务日志**：我们分析的**对象**（"2026-01-15 [ERROR] 登录失败"）
- **工具日志**：logalyzer 自己运行时的记录（"开始解析文件""解析完成：成功 1024 行，失败 3 行"）

工具自身用 logging 模块：

```python
import logging


def setup_logger(verbose: bool = False) -> logging.Logger:
    """配置工具自身日志"""
    logger = logging.getLogger("logalyzer")
    logger.setLevel(logging.DEBUG if verbose else logging.INFO)

    handler = logging.StreamHandler()
    formatter = logging.Formatter(
        "[%(asctime)s] %(levelname)s %(name)s: %(message)s",
        datefmt="%H:%M:%S",
    )
    handler.setFormatter(formatter)
    logger.addHandler(handler)
    return logger


# 使用
logger = setup_logger(verbose=True)
logger.info("开始分析日志文件")
logger.debug(f"正则模式: {LOG_PATTERN.pattern}")
logger.info("解析完成: 成功 1024 行, 失败 3 行")
logger.warning("有 3 行格式无法识别，已跳过")
logger.info("统计完成，正在输出报告")
```

输出：

```

[10:23:45] DEBUG logalyzer: 正则模式: ^(\d{4}-\d{2}-\d{2} ...
```

## 输出层完整代码

```python
import json
import csv
import io
import logging
from collections import Counter, defaultdict


def format_table(entries: list, max_rows: int = 20) -> str:
    """控制台表格输出"""
    lines = []
    lines.append(f"{'时间':<22} {'级别':<8} {'模块':<10} 消息")
    lines.append("-" * 80)
    for e in entries[:max_rows]:
        msg = e.message[:50]
        lines.append(f"{e.timestamp:<22} {e.level:<8} {e.module:<10} {msg}")
    if len(entries) > max_rows:
        lines.append(f"... 共 {len(entries)} 条，仅显示前 {max_rows} 条")
    return "\n".join(lines)


def build_stats(entries: list) -> dict:
    """从 entries 构建统计报告"""
    return {
        "level_counts": dict(Counter(e.level for e in entries)),
        "hourly_counts": dict(sorted(Counter(e.hour for e in entries).items())),
        "top_errors": Counter(
            e.message for e in entries if e.level == "ERROR"
        ).most_common(5),
        "total": len(entries),
    }


def export_json(stats: dict, indent: int = 2) -> str:
    return json.dumps(stats, ensure_ascii=False, indent=indent)


def export_csv(stats: dict) -> str:
    buf = io.StringIO()
    writer = csv.writer(buf)
    writer.writerow(["类型", "名称", "数量"])
    for level, count in stats["level_counts"].items():
        writer.writerow(["级别", level, count])
    for msg, count in stats["top_errors"]:
        writer.writerow(["错误", msg, count])
    return buf.getvalue()


def setup_logger(verbose: bool = False) -> logging.Logger:
    logger = logging.getLogger("logalyzer")
    if logger.handlers:
        return logger  # 避免重复添加 handler
    logger.setLevel(logging.DEBUG if verbose else logging.INFO)
    handler = logging.StreamHandler()
    fmt = logging.Formatter(
        "[%(asctime)s] %(levelname)s %(name)s: %(message)s",
        datefmt="%H:%M:%S",
    )
    handler.setFormatter(fmt)
    logger.addHandler(handler)
    return logger
```

## 翻车现场：JSON 导出中文变成 Unicode 转义

```python
import json

data = {"message": "用户登录失败"}

# WRONG：默认 ensure_ascii=True
print(json.dumps(data))
# 输出: {"message": "\u7528\u6237\u767b\u5f55\u5931\u8d25"}
```

**错误输出**：`{"message": "\u7528\u6237\u767b\u5f55\u5931\u8d25"}`——虽然程序能读，但人完全看不懂，排错时非常痛苦。

**修复**：`json.dumps(data, ensure_ascii=False)` → `{"message": "用户登录失败"}`

## 翻车现场：CSV 用 \n 拼接忘记 Excel 编码

```python
# WRONG：直接字符串拼接
csv_text = "级别,数量\nERROR,8\nINFO,5\n"
with open("report.csv", "w") as f:
    f.write(csv_text)
```

**问题**：Windows 上用 Excel 打开可能中文乱码，而且行尾 `\n` 在某些环境变成 `\r\r\n`。

**修复**：用 `csv` 模块 + 指定编码：

```python
import csv

with open("report.csv", "w", newline="", encoding="utf-8-sig") as f:
    writer = csv.writer(f)
    writer.writerow(["级别", "数量"])
    writer.writerow(["ERROR", 8])
```

`utf-8-sig` 编码会在文件开头写入 BOM，Excel 打开不会乱码。

## 翻车现场：logger 重复添加 handler 导致日志翻倍

```python
import logging

def get_logger():
    logger = logging.getLogger("myapp")
    logger.setLevel(logging.INFO)
    handler = logging.StreamHandler()
    logger.addHandler(handler)  # 每次调用都加一个！
    return logger

log1 = get_logger()
log2 = get_logger()
log1.info("hello")
```

**错误输出**：

```
hello
hello
```

**每条日志打印了两遍！** 因为 `getLogger` 返回同一个 logger 实例，handler 被加了两次。

**修复**：加 handler 前检查：

```python
def get_logger():
    logger = logging.getLogger("myapp")
    if logger.handlers:
        return logger  # 已有 handler，直接返回
    logger.setLevel(logging.INFO)
    handler = logging.StreamHandler()
    logger.addHandler(handler)
    return logger
```

## 小结

- 输出层把统计数据变成人类可读/机器可解析的格式
- 控制台表格适合调试，JSON 适合程序传递，CSV 适合导入 Excel
- JSON 导出务必 `ensure_ascii=False`，否则中文变 Unicode 转义
- CSV 导出用 `csv` 模块 + `utf-8-sig` 编码，兼容 Excel
- 工具自身的运行日志用 logging，和要分析的业务日志是两回事
- 避免 logger 重复添加 handler 导致日志翻倍
