## argparse 子命令组装

回顾第 4 章学的 `argparse`。我们的工具需要三个子命令：`analyze`（分析并展示）、`export`（导出文件）、`stats`（统计摘要）。

```python
import argparse
import json
import logging
import sys
from collections import Counter, defaultdict

from dataclasses import dataclass


@dataclass
class LogEntry:
    timestamp: str
    level: str
    module: str
    message: str

    @property
    def hour(self):
        return self.timestamp.split(" ")[1].split(":")[0]


LOG_PATTERN = re_compiled_pattern = None  # placeholder


def parse_line(line: str) -> LogEntry | None:
    line = line.strip()
    if not line:
        return None
    match = re.match(
        r'^(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}) '
        r'\[(\w+)\] '
        r'\[(\w+)\] '
        r'(.*)$',
        line,
    )
    if match is None:
        return None
    return LogEntry(match.group(1), match.group(2), match.group(3), match.group(4).strip())


def parse_lines(lines):
    entries, failed = [], []
    for line in lines:
        e = parse_line(line)
        if e is None:
            failed.append(line.strip())
        else:
            entries.append(e)
    return entries, failed


def build_stats(entries):
    return {
        "total": len(entries),
        "level_counts": dict(Counter(e.level for e in entries)),
        "hourly_counts": dict(sorted(Counter(e.hour for e in entries).items())),
        "top_errors": Counter(
            e.message for e in entries if e.level == "ERROR"
        ).most_common(5),
        "module_errors": dict(sorted(
            ((m, c) for m, c in defaultdict(int, {}).items()) if False else
            sorted(defaultdict(int, {}).items(), key=lambda x: x[1], reverse=True)
        )),
    }


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="logalyzer",
        description="日志分析 CLI 工具",
        formatter_class=argparse.RawDescriptionHelpFormatter,
    )
    parser.add_argument("--version", action="version", version="logalyzer 0.1.0")
    parser.add_argument("--verbose", "-v", action="store_true",
                        help="显示详细运行日志")

    sub = parser.add_subparsers(dest="command")

    # analyze 子命令
    p_analyze = sub.add_parser("analyze", help="分析日志并输出控制台报告")
    p_analyze.add_argument("input", help="日志文件路径（留空则读 stdin）")
    p_analyze.add_argument("--top", type=int, default=5, help="Top N 错误数")

    # export 子命令
    p_export = sub.add_parser("export", help="导出统计结果为文件")
    p_export.add_argument("input", help="日志文件路径")
    p_export.add_argument("-f", "--format", choices=["json", "csv"],
                          default="json", help="导出格式")
    p_export.add_argument("-o", "--output", help="输出文件路径")

    # stats 子命令
    p_stats = sub.add_parser("stats", help="仅输出统计摘要（JSON）")
    p_stats.add_argument("input", help="日志文件路径")

    return parser
```

## 主流程

```python
def run_analyze(args, logger) -> int:
    """analyze 子命令实现"""
    with open(args.input, encoding="utf-8") as f:
        content = f.read()
    lines = content.splitlines()
    logger.info(f"读取 {len(lines)} 行日志")

    entries, failed = parse_lines(lines)
    logger.info(f"解析完成: 成功 {len(entries)} 行, 失败 {len(failed)} 行")
    if failed:
        logger.warning(f"{len(failed)} 行格式无法识别")

    stats = build_stats(entries)

    # 控制台输出
    print("=" * 50)
    print("📊 日志分析报告")
    print("=" * 50)
    print(f"\n总计: {stats['total']} 条日志\n")

    print("[级别分布]")
    total = stats["total"]
    for level, count in sorted(stats["level_counts"].items(),
                                key=lambda x: x[1], reverse=True):
        ratio = count / total * 100 if total else 0
        print(f"  {level:<8} {count:<6} {ratio:.1f}%")

    print(f"\n[Top {args.top} 错误]")
    for i, (msg, count) in enumerate(stats["top_errors"][:args.top], 1):
        print(f"  {i}. [{count}次] {msg}")

    return 0


def run_export(args, logger) -> int:
    """export 子命令实现"""
    with open(args.input, encoding="utf-8") as f:
        entries, failed = parse_lines(f.read().splitlines())

    stats = build_stats(entries)
    logger.info(f"解析 {len(entries)} 条, 失败 {len(failed)} 条")

    if args.format == "json":
        output = json.dumps(stats, ensure_ascii=False, indent=2)
        ext = "json"
    else:
        import csv, io
        buf = io.StringIO()
        writer = csv.writer(buf)
        writer.writerow(["类型", "名称", "数量"])
        for level, count in stats["level_counts"].items():
            writer.writerow(["级别", level, count])
        output = buf.getvalue()
        ext = "csv"

    if args.output:
        with open(args.output, "w", encoding="utf-8") as f:
            f.write(output)
        logger.info(f"已写入 {args.output}")
    else:
        out_path = f"report.{ext}"
        with open(out_path, "w", encoding="utf-8") as f:
            f.write(output)
        logger.info(f"已写入 {out_path}")

    return 0


def run_stats(args, logger) -> int:
    """stats 子命令：仅输出 JSON 摘要"""
    with open(args.input, encoding="utf-8") as f:
        entries, _ = parse_lines(f.read().splitlines())

    stats = build_stats(entries)
    print(json.dumps({
        "total": stats["total"],
        "level_counts": stats["level_counts"],
        "top_errors": stats["top_errors"],
    }, ensure_ascii=False, indent=2))
    return 0


def main(argv=None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)

    logger = logging.getLogger("logalyzer")
    level = logging.DEBUG if getattr(args, "verbose", False) else logging.INFO
    logging.basicConfig(level=level, format="[%(levelname)s] %(message)s")

    if args.command is None:
        parser.print_help()
        return 0

    handlers = {
        "analyze": run_analyze,
        "export": run_export,
        "stats": run_stats,
    }
    handler = handlers.get(args.command)
    if handler is None:
        parser.error(f"未知子命令: {args.command}")
        return 1

    return handler(args, logger)


if __name__ == "__main__":
    sys.exit(main())
```

## 使用方式

```bash

```

```python
import unittest
from logalyzer.parser import parse_line, parse_lines
from logalyzer.stats import (
    level_distribution, hourly_distribution,
    top_errors, module_error_count,
)


class TestParser(unittest.TestCase):
    def test_parse_valid_line(self):
        line = "2026-01-15 10:23:45 [ERROR] [auth] 登录失败"
        entry = parse_line(line)
        self.assertIsNotNone(entry)
        self.assertEqual(entry.level, "ERROR")
        self.assertEqual(entry.module, "auth")
        self.assertEqual(entry.message, "登录失败")

    def test_parse_invalid_line(self):
        self.assertIsNone(parse_line("这行格式不对"))
        self.assertIsNone(parse_line(""))
        self.assertIsNone(parse_line("   "))

    def test_parse_lines_partial(self):
        lines = [
            "2026-01-15 10:23:45 [INFO] [order] ok",
            "垃圾行",
            "2026-01-15 10:23:46 [ERROR] [auth] fail",
            "",
        ]
        entries, failed = parse_lines(lines)
        self.assertEqual(len(entries), 2)
        self.assertEqual(len(failed), 2)
        self.assertEqual(entries[0].level, "INFO")
        self.assertEqual(entries[1].level, "ERROR")


class TestStats(unittest.TestCase):
    def setUp(self):
        from dataclasses import dataclass

        @dataclass
        class FakeEntry:
            pass

        self.entries = [
```

## 翻车现场：子命令参数解析报错"too few arguments"

```python
# WRONG：忘记设置 required=True 或者 dest
sub = parser.add_subparsers()

# 但用户只输入 `logalyzer analyze` 不带文件路径
```

**错误输出**：

```
usage: logalyzer analyze [-h] input
logalyzer analyze: error: the following arguments are required: input
```

**这是预期行为**——但如果你希望 `input` 可选（从 stdin 读），需要设 `nargs="?"`：

```python
p_analyze.add_argument("input", nargs="?", default=None,
                       help="日志文件路径（留空则读 stdin）")
```

## 翻车现场：main() 里直接 open 文件，单元测试无法覆盖

```python
# WRONG：直接在主流程里读文件
def run_analyze(args):
    with open(args.input) as f:       # 测试时必须要有真实文件
        entries, _ = parse_lines(f.read().splitlines())
    ...
```

**问题**：单元测试要覆盖 `run_analyze`，就必须创建临时文件，测试又慢又脆弱。

**修复**：把"读文件"和"处理逻辑"分开，让处理逻辑接收字符串列表：

```python
def analyze_logic(lines: list[str]) -> dict:
    """纯逻辑，不碰文件，可直接单测"""
    entries, failed = parse_lines(lines)
    return build_stats(entries)

def run_analyze(args):
    with open(args.input) as f:
        return analyze_logic(f.read().splitlines())  # 薄薄一层 IO
```

## 三本书带走了什么

回顾这三本《Python 标准库》：

**《标准库 1》—— 数据与文件**
- 文本处理：str / re / difflib / textwrap
- 数据结构：list / dict / set / collections
- 文件与路径：pathlib / os / shutil / csv / json
- 数据压缩：gzip / zipfile / tarfile

**《标准库 2》—— 并发与时间**
- 时间处理：datetime / time / calendar / zoneinfo
- 并发编程：threading / multiprocessing / asyncio / concurrent.futures
- 网络基础：socket / urllib / http.server
- 进程与系统：subprocess / os / platform

**《标准库 3》—— 文本、工具与工程化**
- 正则引擎：re 模块的编译、分组、贪婪与回溯
- 效率工具：collections 高级用法、itertools、functools
- 日志与 CLI：logging 五级别、argparse 子命令
- 测试与质量：unittest / pytest、测试组织与覆盖率

**三本书合起来，覆盖了多少？** Python 标准库有 200+ 个模块，我们只深入了约 40 个核心模块。但**这 40 个模块覆盖了 80% 的日常开发场景**。剩下的模块，你现在已经知道怎么查文档、怎么上手了。

## 小结

- argparse 子命令把解析/统计/输出三层串联成完整 CLI 工具
- 子命令用 `add_subparsers()` + `add_parser()` 注册，各自有独立参数
- 主流程函数保持"薄"：只做 IO，逻辑交给纯函数（方便测试）
- unittest 可以测试解析层和统计层，每一层独立覆盖
- 三本标准库覆盖了 40 个核心模块，足以应对 80% 场景
- 下一步按方向选择：数据分析 / 算法竞赛 / 桌面应用 / Web 工程
