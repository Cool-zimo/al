# 实战：给一个小模块写完整测试

> 这一课把前面所有知识串起来：写一个真实的小模块（购物车），写完整的单元测试，跑通，然后故意引入 bug 看测试能否抓到。

## 被测模块：购物车

这是一个简化但完整的购物车模块，包含商品管理、金额计算、优惠券逻辑：

```python
# cart.py
class Cart:
    def __init__(self):
        self.items = {}       # {sku: {"name": ..., "price": ..., "qty": ...}}
        self.coupon = None    # 优惠券

    def add(self, sku, name, price, qty=1):
        if price < 0:
            raise ValueError("价格不能为负")
        if qty <= 0:
            raise ValueError("数量必须大于 0")
        if sku in self.items:
            self.items[sku]["qty"] += qty
        else:
            self.items[sku] = {"name": name, "price": price, "qty": qty}

    def remove(self, sku):
        if sku not in self.items:
            raise KeyError(f"商品 {sku} 不在购物车中")
        del self.items[sku]

    def total(self):
        return round(sum(
            item["price"] * item["qty"]
            for item in self.items.values()
        ), 2)

    def apply_coupon(self, coupon_type, value):
        """coupon_type: 'percent' 百分比折扣, 'fixed' 固定减免"""
        if coupon_type not in ("percent", "fixed"):
            raise ValueError("不支持的优惠券类型")
        self.coupon = {"type": coupon_type, "value": value}

    def final_price(self):
        t = self.total()
        if self.coupon is None:
            return t
        if self.coupon["type"] == "percent":
            return round(t * (1 - self.coupon["value"] / 100), 2)
        elif self.coupon["type"] == "fixed":
            return round(max(0, t - self.coupon["value"]), 2)
        return t

    def is_empty(self):
        return len(self.items) == 0
```

## 测试用例

```python
# test_cart.py
import unittest
from cart import Cart

class TestCart(unittest.TestCase):
    def setUp(self):
        self.cart = Cart()

    def test_empty_cart(self):
        self.assertTrue(self.cart.is_empty())
        self.assertEqual(self.cart.total(), 0)

    def test_add_item(self):
        self.cart.add("a001", "苹果", 5.0, 2)
        self.assertFalse(self.cart.is_empty())
        self.assertEqual(self.cart.total(), 10.0)

    def test_add_duplicate_item(self):
        self.cart.add("a001", "苹果", 5.0, 2)
        self.cart.add("a001", "苹果", 5.0, 3)
        self.assertEqual(self.cart.total(), 25.0)   # 5 * 5

    def test_add_negative_price(self):
        with self.assertRaises(ValueError):
            self.cart.add("a001", "苹果", -5.0)

    def test_add_zero_qty(self):
        with self.assertRaises(ValueError):
            self.cart.add("a001", "苹果", 5.0, 0)

    def test_remove_item(self):
        self.cart.add("a001", "苹果", 5.0, 2)
        self.cart.remove("a001")
        self.assertTrue(self.cart.is_empty())

    def test_remove_nonexistent(self):
        with self.assertRaises(KeyError):
            self.cart.remove("ghost")

    def test_percent_coupon(self):
        self.cart.add("a001", "苹果", 100.0, 2)
        self.cart.apply_coupon("percent", 20)   # 8 折
        self.assertEqual(self.cart.final_price(), 160.0)

    def test_fixed_coupon(self):
        self.cart.add("a001", "苹果", 100.0)
        self.cart.apply_coupon("fixed", 30)
        self.assertEqual(self.cart.final_price(), 70.0)

    def test_fixed_coupon_exceeds_total(self):
        self.cart.add("a001", "苹果", 10.0)
        self.cart.apply_coupon("fixed", 30)   # 减免超过总价
        self.assertEqual(self.cart.final_price(), 0)   # 不低于 0

    def test_invalid_coupon_type(self):
        with self.assertRaises(ValueError):
            self.cart.apply_coupon("shipping", 10)

    def test_multiple_items(self):
        self.cart.add("a001", "苹果", 5.0, 2)
        self.cart.add("b002", "香蕉", 3.0, 5)
        self.assertEqual(self.cart.total(), 25.0)   # 10 + 15

    def test_float_precision(self):
        self.cart.add("a001", "咖啡", 3.33, 3)
        self.assertAlmostEqual(self.cart.total(), 9.99, places=2)

if __name__ == "__main__":
    unittest.main()
```

**13 个测试，覆盖**：空购物车、添加、重复添加、参数校验、删除、不存在的商品、百分比优惠券、固定优惠券、减免超额、非法优惠券、多商品、浮点精度。

## 故意引入 bug，看测试抓到

现在我们把 `final_price` 里的百分比计算改错：

```python
    def final_price(self):
        t = self.total()
        if self.coupon is None:
            return t
        if self.coupon["type"] == "percent":
            return round(t * self.coupon["value"] / 100, 2)  # BUG：忘了 1-
        ...
```

**原来是 `t * (1 - value/100)`，改成了 `t * value/100`**（把"打 8 折"算成了"乘以 20%"）。

再跑测试：

```
test_percent_coupon (__main__.TestCart) ... FAIL
...
AssertionError: 40.0 != 160.0
```

**`test_percent_coupon` 立刻报错**——20% 折扣算出 40 而不是 160，和预期不符。测试抓住了 bug。

再试另一个 bug：优惠券减免超过总价时忘记 `max(0, ...)`：

```python
    def final_price(self):
        ...
        elif self.coupon["type"] == "fixed":
            return t - self.coupon["value"]   # BUG：可能是负数
```

跑测试：

```
test_fixed_coupon_exceeds_total (__main__.TestCart) ... FAIL
...
AssertionError: -20 != 0
```

**`test_fixed_coupon_exceeds_total` 抓到了**——价格算成了 -20，不符合"不低于 0"的预期。

## 翻车现场：只测正常路径，bug 溜进生产

```python
# 只测了 happy path 的测试
class TestCartWeak(unittest.TestCase):
    def test_add(self):
        cart = Cart()
        cart.add("a001", "苹果", 5.0, 2)
        self.assertEqual(cart.total(), 10.0)

    def test_total(self):
        cart = Cart()
        cart.add("a001", "苹果", 100.0, 2)
        cart.apply_coupon("percent", 20)
        self.assertEqual(cart.final_price(), 160.0)
```

**只有 2 个测试，全是正例。** 如果有人改了 `final_price` 的百分比逻辑，第一个测试发现不了（它没用优惠券），只有第二个能抓到。但如果 bug 在 `add` 的参数校验里（比如允许负数价格），两个测试都发现不了——因为没有测异常路径。

**测试的正例和反例缺一不可。**

## 翻车现场：测试和被测试代码同步出错

```python
# cart.py（有 bug）
def total(self):
    return round(sum(item["price"] for item in self.items.values()), 2)  # 忘了 * qty

# test_cart.py（期望值跟着错了）
def test_add_item(self):
    self.cart.add("a001", "苹果", 5.0, 2)
    self.assertEqual(self.cart.total(), 5.0)   # 期望 5 而不是 10
```

**测试通过了，但 bug 还在**——因为测试的期望值抄了错误实现的输出。这是"测试和被测试代码同步出错"的经典陷阱。避免方法：测试数据手算、多写边界用例、让另一个人 review 测试。

## 小结

- 一个完整模块要测：正常路径、边界值、异常路径、参数校验
- 测试用例要覆盖每个方法的核心逻辑和错误处理
- 故意引入 bug 验证测试能抓到，是检验测试质量的有效手段
- 只测正常路径的测试是"安慰剂测试"，bug 照样溜走
- 测试期望值要手算，不能抄代码输出，否则同步出错
- 13 个测试跑 0.001 秒，比手动测一遍快了几万倍
