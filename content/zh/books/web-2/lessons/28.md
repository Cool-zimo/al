# 实现增删改查：渲染函数与核心交互

> 这一节解决"怎么把上一节的数据结构变成可交互的待办应用"的问题——实现渲染函数和增删改查逻辑。

## 完整代码骨架

```javascript
// ====== 数据 ======
let todos = [];
let nextId = 1;

// ====== DOM 引用 ======
const todoInput = document.querySelector('#todo-input');
const addBtn = document.querySelector('#add-btn');
const todoList = document.querySelector('#todo-list');
const statsEl = document.querySelector('#stats');

// ====== 渲染函数 ======
function render() {
  // 清空列表
  todoList.innerHTML = '';

  // 遍历渲染每个 todo
  todos.forEach(todo => {
    const li = document.createElement('li');
    if (todo.done) li.classList.add('done');

    // 复选框
    const checkbox = document.createElement('input');
    checkbox.type = 'checkbox';
    checkbox.checked = todo.done;
    checkbox.addEventListener('click', () => toggleTodo(todo.id));

    // 文本
    const span = document.createElement('span');
    span.textContent = todo.text;

    // 删除按钮
    const deleteBtn = document.createElement('button');
    deleteBtn.textContent = '删除';
    deleteBtn.className = 'delete-btn';
    deleteBtn.addEventListener('click', () => deleteTodo(todo.id));

    // 组装
    li.appendChild(checkbox);
    li.appendChild(span);
    li.appendChild(deleteBtn);
    todoList.appendChild(li);
  });

  // 更新统计
  updateStats();
}

// ====== 添加 ======
function addTodo() {
  const text = todoInput.value.trim();
  if (text === '') {
    alert('任务内容不能为空');
    return;
  }

  todos.push({ id: nextId++, text, done: false });
  todoInput.value = '';  // 清空输入框
  render();              // 重新渲染
}

// ====== 删除 ======
function deleteTodo(id) {
  todos = todos.filter(todo => todo.id !== id);
  render();
}

// ====== 切换完成 ======
function toggleTodo(id) {
  const todo = todos.find(todo => todo.id === id);
  if (todo) {
    todo.done = !todo.done;
    render();
  }
}

// ====== 统计 ======
function updateStats() {
  const total = todos.length;
  const done = todos.filter(t => t.done).length;
  statsEl.textContent = `共 ${total} 项，${done} 项已完成`;
}

// ====== 事件绑定 ======
addBtn.addEventListener('click', addTodo);

todoInput.addEventListener('keydown', (event) => {
  if (event.key === 'Enter') {
    addTodo();
  }
});
```

## 用事件委托优化

上面的代码给每个 checkbox 和 button 都绑了监听器。用事件委托优化：

```javascript
// 在 todoList 上统一监听
todoList.addEventListener('click', (event) => {
  const li = event.target.closest('li');
  if (!li) return;

  const id = Number(li.dataset.id);

  // 判断点击的是删除按钮
  if (event.target.classList.contains('delete-btn')) {
    deleteTodo(id);
    return;
  }

  // 判断点击的是复选框
  if (event.target.type === 'checkbox') {
    toggleTodo(id);
    return;
  }
});
```

翻车现场一：渲染时没把 id 存到 DOM 上。

```javascript
// ❌ 渲染时没有把 id 存到 li 上
function render() {
  todoList.innerHTML = '';
  todos.forEach(todo => {
    const li = document.createElement('li');
    li.textContent = todo.text;  // 没有存 id！
    todoList.appendChild(li);
  });
}

// 后面事件委托时拿不到 id
todoList.addEventListener('click', (event) => {
  const li = event.target.closest('li');
  const id = li.dataset.id;  // undefined！因为渲染时没设 data-id
});
```

后果：事件委托时无法从 DOM 上获取 todo 的 id，无法定位要操作的数据对象。

```javascript
// ✅ 渲染时把 id 存到 data-id 上
function render() {
  todoList.innerHTML = '';
  todos.forEach(todo => {
    const li = document.createElement('li');
    li.dataset.id = todo.id;  // 关键！把 id 存到 DOM
    li.className = todo.done ? 'done' : '';
    li.innerHTML = `
      <input type="checkbox" ${todo.done ? 'checked' : ''}>
      <span>${todo.text}</span>
      <button class="delete-btn">删除</button>
    `;
    todoList.appendChild(li);
  });
  updateStats();
}
```

## 模板字符串 vs createElement

两种渲染方式对比：

```javascript
// 方式一：innerHTML + 模板字符串（简洁但注意 XSS）
li.innerHTML = `
  <input type="checkbox" ${todo.done ? 'checked' : ''}>
  <span>${todo.text}</span>
  <button class="delete-btn">删除</button>
`;

// 方式二：createElement（安全但冗长）
const checkbox = document.createElement('input');
checkbox.type = 'checkbox';
// ... 一堆 appendChild
```

**安全提醒**：如果 `todo.text` 来自用户输入，`innerHTML` 有 XSS 风险。用 `textContent` 或先转义。

```javascript
// 安全的模板渲染：对用户输入做转义
function escapeHtml(str) {
  return str.replace(/[&<>"']/g, char => ({
    '&': '&amp;',
    '<': '&lt;',
    '>': '&gt;',
    '"': '&quot;',
    "'": '&#39;'
  }[char]));
}

li.innerHTML = `
  <span>${escapeHtml(todo.text)}</span>
`;
```

翻车现场二：innerHTML 注入攻击。

```javascript
// 如果用户输入了恶意内容
todos.push({ id: 1, text: '<img src=x onerror="alert(1)">', done: false });

// 用 innerHTML 渲染时，脚本会执行！
li.innerHTML = `<span>${todo.text}</span>`;
// 弹出 alert(1) —— XSS 攻击成功
```

后果：恶意用户可以通过输入脚本代码攻击其他用户。永远对用户输入做转义，或者用 `textContent` 代替 `innerHTML`。

## 添加功能的边界处理

```javascript
function addTodo() {
  const text = todoInput.value.trim();

  // 边界1：空字符串
  if (text === '') {
    alert('请输入任务内容');
    todoInput.focus();
    return;
  }

  // 边界2：超长文本
  if (text.length > 100) {
    alert('任务内容不能超过100字');
    return;
  }

  // 边界3：重复任务
  if (todos.some(t => t.text === text)) {
    alert('该任务已存在');
    return;
  }

  todos.push({ id: nextId++, text, done: false });
  todoInput.value = '';
  todoInput.focus();  // 添加后保持焦点，方便连续输入
  render();
}
```

## 与 Python 的对比

Python 中的列表操作：

```python
# Python：增删改查
todos = []

# 增
todos.append({"id": 1, "text": "买牛奶", "done": False})

# 删
todos = [t for t in todos if t["id"] != 1]

# 改
for t in todos:
    if t["id"] == 1:
        t["done"] = True

# 查
for t in todos:
    print(t["text"])
```

JS 版本的数组操作（`push`、`filter`、`find`）与 Python 几乎一一对应，只是多了"改完数据要重新渲染 DOM"这一步。

## 小结

- 渲染函数是核心：清空 DOM → 遍历数据 → 创建元素 → 组装 → 更新统计。
- 事件委托优化：在父元素上统一监听，用 `data-id` 关联 DOM 和数据。
- 渲染时**必须**把 id 存到 DOM 上（data-id），否则委托时拿不到。
- `innerHTML` 有 XSS 风险，用户输入必须转义或用 `textContent`。
- 添加功能要做边界处理：空值、超长、重复。
- 数据变化后必须调用 `render()` 更新视图。
