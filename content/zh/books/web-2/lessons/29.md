# 加持久化与筛选：localStorage 存取与条件渲染

> 这一节解决"怎么让数据刷新不丢失，以及怎么按条件筛选显示"的问题。

## 持久化：localStorage 读写

```javascript
const STORAGE_KEY = 'todos';

// 从 localStorage 读取
function loadTodos() {
  const raw = localStorage.getItem(STORAGE_KEY);
  if (raw === null) return [];  // 首次访问，返回空数组
  try {
    return JSON.parse(raw);
  } catch (err) {
    console.error('解析 todos 失败：', err);
    return [];  // 数据损坏，返回空数组
  }
}

// 保存到 localStorage
function saveTodos() {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));
}

// 初始化
let todos = loadTodos();
let nextId = todos.length > 0
  ? Math.max(...todos.map(t => t.id)) + 1  // 取最大 id + 1
  : 1;
```

**注意 nextId 的恢复**：页面刷新后 `nextId` 要恢复为当前最大 id + 1，否则可能和已有 id 冲突。

翻车现场一：刷新后 nextId 从 1 开始，导致 id 冲突。

```javascript
// ❌ 刷新后 nextId 重置为 1
let nextId = 1;

// 已有 todos: [{ id: 1, text: '买牛奶' }, { id: 2, text: '写代码' }]
// 添加新任务 → id 又是 1 → 和已有任务 id 冲突！
todos.push({ id: nextId++, text: '看书' });  // id = 1，重复了！
```

后果：id 冲突导致后续操作（删除、切换）可能操作到错误的任务。

```javascript
// ✅ 从已有数据中恢复 nextId
let nextId = todos.length > 0
  ? Math.max(...todos.map(t => t.id)) + 1
  : 1;
```

## 改造增删改：加上持久化

```javascript
function addTodo() {
  const text = todoInput.value.trim();
  if (text === '') return;

  todos.push({ id: nextId++, text, done: false });
  todoInput.value = '';
  saveTodos();  // 保存
  render();
}

function deleteTodo(id) {
  todos = todos.filter(todo => todo.id !== id);
  saveTodos();  // 保存
  render();
}

function toggleTodo(id) {
  const todo = todos.find(todo => todo.id === id);
  if (todo) {
    todo.done = !todo.done;
    saveTodos();  // 保存
    render();
  }
}
```

**模式：每次数据变化后都 `saveTodos()` + `render()`。**

## 筛选功能

```javascript
let currentFilter = 'all';  // 'all' | 'active' | 'completed'

// 筛选按钮事件
document.querySelectorAll('.filter-btn').forEach(btn => {
  btn.addEventListener('click', () => {
    // 更新按钮状态
    document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');

    // 更新筛选条件
    currentFilter = btn.dataset.filter;
    render();  // 重新渲染
  });
});

// 改造 render：先过滤再渲染
function render() {
  todoList.innerHTML = '';

  // 根据筛选条件过滤
  const filtered = todos.filter(todo => {
    if (currentFilter === 'active') return !todo.done;
    if (currentFilter === 'completed') return todo.done;
    return true;  // 'all'
  });

  // 渲染过滤后的列表
  filtered.forEach(todo => {
    const li = document.createElement('li');
    li.dataset.id = todo.id;
    li.className = todo.done ? 'done' : '';
    li.innerHTML = `
      <input type="checkbox" ${todo.done ? 'checked' : ''}>
      <span>${escapeHtml(todo.text)}</span>
      <button class="delete-btn">删除</button>
    `;
    todoList.appendChild(li);
  });

  updateStats();
}
```

翻车现场二：筛选后删除，用的是 filtered 数组的下标。

```javascript
// ❌ 如果用 innerHTML 字符串拼接 + 下标
function render() {
  todoList.innerHTML = filtered.map((todo, index) => `
    <li data-index="${index}">
      <span>${todo.text}</span>
      <button onclick="deleteTodo(${index})">删除</button>
    </li>
  `).join('');
}

// 筛选后 index 不等于原始 todos 中的下标！
// "已完成"筛选下，index 0 可能对应原始 todos 的第 2 项
```

后果：用 filtered 数组的下标去操作原始 todos 数组，会操作错对象。必须用 id，不是下标。

## 完整流程图

```
页面加载
  ↓
loadTodos() → 从 localStorage 读取
  ↓
render() → 渲染列表
  ↓
用户操作
  ├── 添加 → addTodo() → todos.push → saveTodos → render
  ├── 删除 → deleteTodo() → todos.filter → saveTodos → render
  ├── 切换 → toggleTodo() → 修改 done → saveTodos → render
  └── 筛选 → 修改 currentFilter → render
```


## 与 Python 的对比

Python 中持久化通常直接读写文件：

```python
import json

def load_todos():
    try:
        with open('todos.json', 'r') as f:
            return json.load(f)
    except FileNotFoundError:
        return []

def save_todos(todos):
    with open('todos.json', 'w') as f:
        json.dump(todos, f)
```

JS 的 localStorage 更方便——自动绑定域名、自动序列化（虽然对象要手动 JSON.stringify）。但容量限制 5MB，不适合存大量数据。

## 小结

- 持久化三件套：`loadTodos()`、`saveTodos()`、每次数据变化后调用 `saveTodos()`。
- 刷新后必须恢复 `nextId` 为最大 id + 1，否则 id 冲突。
- 筛选是渲染层面的事：先 `filter()` 再渲染，不修改原始数据。
- 操作数据始终用 id，不用下标（筛选后下标和原始数据不对应）。
- 核心模式：数据变化 → saveTodos → render，三步一体。
