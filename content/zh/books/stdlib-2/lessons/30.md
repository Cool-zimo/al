## 用 sys.argv 做命令行

《标准库 3》才讲 `argparse`。这一课只用 `sys.argv`，靠自己解析，够用且直观。

```python
import sys
from pathlib import Path


def parse_args(argv):
    """解析命令行参数，返回 dict。
    用法: organize.py <目录> [--apply] [--undo [run_id]] [--rules 规则文件]"""
    args = {"root": None, "apply": False, "undo": None, "rules": None}
    i = 1
    while i < len(argv):
        a = argv[i]
        if a == "--apply":
            args["apply"] = True
        elif a == "--undo":
            args["undo"] = "latest"
            if i + 1 < len(argv) and not argv[i + 1].startswith("--"):
                args["undo"] = argv[i + 1]
                i += 1
        elif a == "--rules":
            args["rules"] = argv[i + 1]
            i += 1
        elif not a.startswith("--"):
            args["root"] = a
        i += 1
    return args


def main(argv):
    args = parse_args(argv)
    if not args["root"]:
        print("用法: python organize.py <目录> [--apply] [--undo [run_id]]")
        return 1

    root = Path(args["root"])
    if not root.exists():
        print(f"错误：目录不存在 -> {root}")
        return 2

    # 四层串联
    from organizer import Organizer   # 假设前面几课的实现放在 organizer.py
    org = Organizer(root)

    if args["undo"]:
        n, errs = org.undo(args["undo"] if args["undo"] != "latest" else None)
        print(f"✓ 已撤销 {n} 个文件" + (f"，{len(errs)} 个失败" if errs else ""))
        return 0

    org.scan()
    org.classify()
    org.plan_moves()
    plan = org.get_plan()

    if not args["apply"]:
        org.preview()
        return 0

    # 真正执行
    ok, failed, run_id = org.apply(plan)
    print(f"✓ 完成 {ok} 个，失败 {len(failed)} 个")
    print(f"  run_id={run_id}（可用 --undo {run_id} 撤销）")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

## 进度反馈：别让用户对着空白屏干等

扫 3000 个文件、逐个 `stat()` 可能要几秒。完全没反馈，用户会以为程序卡死然后 Ctrl-C。

```python
def scan_with_progress(root, ignore_dirs, ignore_files, callback=None):
    """遍历目录并收集元信息，每处理 100 个文件回调一次进度"""
    from datetime import datetime
    root = Path(root)
    files, count = [], 0
    for p in root.rglob("*"):
        if not p.is_file():
            continue
        if any(part in ignore_dirs for part in p.relative_to(root).parts[:-1]):
            continue
        if p.name in ignore_files:
            continue
        try:
            st = p.stat()
        except OSError:
            continue
        files.append({
            "src": str(p),
            "name": p.name,
            "ext": p.suffix.lower(),
            "size": st.st_size,
            "mtime": datetime.fromtimestamp(st.st_mtime).isoformat(timespec="seconds"),
        })
        count += 1
        if callback and count % 100 == 0:
            callback(count)
    return files


def default_callback(n):
    print(f"  已扫描 {n} 个文件...", flush=True)

# 使用
files = scan_with_progress("/d", {"__pycache__", ".git"}, {".DS_Store"}, default_callback)
#   已扫描 100 个文件...
#   已扫描 200 个文件...
```

`flush=True` 很关键：输出默认行缓冲，不加它进度可能攒到最后一起刷出来，等于没有反馈。

## 异常处理：单点失败不扩散

第 27、29 课都提到过这个原则，现在写完整版。一个坏文件、一次权限失败，都不该拖垮整次整理。

```python
import shutil
from pathlib import Path

def move_all(plan, conflict="suffix"):
    """执行整份计划，记录成功与失败，绝不因单点失败中断"""
    ok, failed = 0, []
    for dest_rel, src in plan.items():
        try:
            d = Path(dest_rel)
            if d.exists():
                if conflict == "skip":
                    failed.append({"src": src, "reason": "exists"})
                    continue
                # suffix 策略：自动改名
                stem, suffix = d.stem, d.suffix
                i = 1
                while d.exists():
                    d = d.with_name(f"{stem}_{i}{suffix}")
                    i += 1
            d.parent.mkdir(parents=True, exist_ok=True)
            shutil.move(src, str(d))
            ok += 1
        except OSError as e:
            failed.append({"src": src, "reason": str(e)})
            continue   # 关键：continue 而不是 raise
    return ok, failed


ok, failed = move_all(plan)
print(f"✓ 完成 {ok} 个")
for f in failed:
    print(f"  ✗ 跳过 {f['src']}：{f['reason']}")
```

**三层防护**：`scan` 里 `stat()` 失败跳过、`plan` 里目标冲突有策略、`move` 里 IO 异常捕获并继续。任何一层都不把异常抛给上一层。

## 四层串联：一份能跑的完整骨架

把前面四课的实现放进一个 `organizer.py`，主程序只需十几行就能跑通：

```python
# organizer.py（串联骨架，省略部分实现见前几课）
import json, shutil
from datetime import datetime
from pathlib import Path

IGNORE_DIRS = {"__pycache__", ".git", "node_modules", ".venv", "venv"}
IGNORE_FILES = {".DS_Store", "Thumbs.db"}

DEFAULT_RULES = {
    "categories": {
        "Images": [".jpg", ".jpeg", ".png", ".gif", ".webp"],
        "Docs": [".pdf", ".docx", ".xlsx", ".txt", ".md"],
        "Media": [".mp4", ".mkv", ".mp3"],
        "Archives": [".zip", ".tar", ".gz", ".7z"],
        "Code": [".py", ".js", ".ts", ".go"],
    },
    "date_archive": True,
    "date_format": "%Y-%m",
    "size_thresholds": {"tiny": 10240, "huge": 104857600},
    "conflict": "suffix",
}


class Organizer:
    def __init__(self, root, rules=None, log_dir=None):
        self.root = Path(root)
        self.rules = rules or DEFAULT_RULES
        self.log_dir = Path(log_dir or self.root / ".organize_logs")
        self.files = []
        self.plan = {}

    def scan(self, callback=None):
        self.files = scan_with_progress(self.root, IGNORE_DIRS, IGNORE_FILES, callback)

    def classify(self):
        ext_map = self._build_ext_map()
        for f in self.files:
            f["category"] = ext_map.get(f["ext"]) or self._by_size(f["size"])

    def plan_moves(self):
        for f in self.files:
            cat = f["category"]
            if self.rules["date_archive"]:
                sub = datetime.fromisoformat(f["mtime"]).strftime(self.rules["date_format"])
                dest = f"{cat}/{sub}/{f['name']}"
            else:
                dest = f"{cat}/{f['name']}"
            self.plan[dest] = f["src"]

    def preview(self):
        by_cat = {}
        for dest in self.plan:
            cat = dest.split("/")[0]
            by_cat[cat] = by_cat.get(cat, 0) + 1
        print(f"[预览] 将处理 {len(self.plan)} 个文件")
        for cat, n in sorted(by_cat.items(), key=lambda x: -x[1]):
            print(f"  {cat:<12} -> {n} 个")
        print("\n提示：加 --apply 才会真正执行。")

    def apply(self):
        self.log_dir.mkdir(parents=True, exist_ok=True)
        run_id = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
        log = {"run_id": run_id, "root": str(self.root),
               "operations": [{"src": s, "dest": str(self.root / d), "status": "pending"}
                              for d, s in self.plan.items()]}
        log_file = self.log_dir / f"ops_{run_id}.json"
        log_file.write_text(json.dumps(log, ensure_ascii=False, indent=2))

        ok, failed = 0, []
        for op in log["operations"]:
            try:
                d = Path(op["dest"])
                d.parent.mkdir(parents=True, exist_ok=True)
                shutil.move(op["src"], str(d))
                op["status"] = "done"
                ok += 1
            except OSError as e:
                op["status"] = "failed"
                op["error"] = str(e)
                failed.append(op["src"])
        log_file.write_text(json.dumps(log, ensure_ascii=False, indent=2))
        return ok, failed, run_id

    def undo(self, run_id=None):
        d = self.log_dir
        if run_id:
            log_file = d / f"ops_{run_id}.json"
        else:
            logs = sorted(d.glob("ops_*.json"))
            if not logs:
                raise FileNotFoundError("没有可撤销的操作记录")
            log_file = logs[-1]
        log = json.loads(log_file.read_text(encoding="utf-8"))
        undone = 0
        for op in reversed(log["operations"]):
            if op["status"] != "done":
                continue
            try:
                src, dst = Path(op["dest"]), Path(op["src"])
                if src.exists():
                    dst.parent.mkdir(parents=True, exist_ok=True)
                    shutil.move(str(src), str(dst))
                    undone += 1
            except OSError:
                continue
        return undone

    def get_plan(self):
        return self.plan

    def _build_ext_map(self):
        m = {}
        for cat, exts in self.rules["categories"].items():
            for ext in exts:
                m[ext.lower()] = cat
        return m

    def _by_size(self, size):
        t = self.rules["size_thresholds"]
        if size < t["tiny"]:
            return "Tiny"
        if size >= t["huge"]:
            return "Huge"
        return "其他"
```

## 翻车现场：在 __main__ 里写死逻辑，无法复用

把扫描、分类、执行全塞进 `if __name__ == "__main__":` 里，是新手最常见的写法——能跑，但没法导入、没法测试。

```python
# WRONG：所有逻辑都在 __main__ 里
if __name__ == "__main__":
    root = sys.argv[1]
    for p in Path(root).rglob("*"):
        ...   # 扫描、分类、移动全混在一起
    print("done")
# 别的模块想调用？做不到。写测试？做不到。

# 正确：把逻辑放进类或函数，__main__ 只做"接线"
if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

`if __name__ == "__main__":` 的含义是"只在直接运行这个文件时才执行"。被 `import` 时里面的代码不跑——这让同一个文件既能当程序运行，又能被测试导入。

## 翻车现场：参数解析漏掉边界情况

自己写参数解析时，最容易在"最后一个参数"或"未知选项"上出错。

```python
import sys

# WRONG：不检查越界，--undo 后面没参数会 IndexError
def parse_wrong(argv):
    args = {}
    i = 1
    while i < len(argv):
        if argv[i] == "--undo":
            args["undo"] = argv[i + 1]   # 如果 --undo 是最后一个参数，崩
        i += 1

# 正确：用 i+1 前先判断边界
def parse_right(argv):
    args = {"undo": None}
    i = 1
    while i < len(argv):
        a = argv[i]
        if a == "--undo":
            if i + 1 < len(argv) and not argv[i + 1].startswith("--"):
                args["undo"] = argv[i + 1]
                i += 1
            else:
                args["undo"] = "latest"
        i += 1
    return args

print(parse_right(["organize.py", "--undo"]))            # {'undo': 'latest'}
print(parse_right(["organize.py", "--undo", "abc123"]))  # {'undo': 'abc123'}
```

## 这本书带走了什么

五章走完，你手上攒了一套"和操作系统打交道"的工具箱：

| 能力 | 标准库 | 一句话 |
|---|---|---|
| 时间计算 | `datetime`/`time`/`calendar` | 时间就是数字，时区要显式，格式要统一 |
| 路径与遍历 | `pathlib`/`os.path` | 新代码用 `pathlib`，路径是对象不是字符串 |
| 文件读写 | `open`/`io`/`tempfile` | `with` 保命，临时文件交给 `tempfile` |
| 系统交互 | `os`/`subprocess`/`shutil` | 子进程用列表传参，别拼 shell 字符串 |
| 归档压缩 | `zipfile`/`tarfile` | 压缩比不是玄学，文本可压、媒体难压 |

更重要的是三个**工程习惯**：

1. **分层**：扫描、分类、规则、执行各管一段，纯函数在前，IO 在后——可测试、可复用
2. **防御**：先预览再执行、先写日志再动手、单点失败不扩散
3. **配置化**：规则是数据不是代码，换个场景不用改逻辑

这本书解决的是"把事情做对"。下一本书解决的是"把代码写好、好维护"——

## 小结

- `sys.argv` 手动解析够用，注意越界和"选项后无参数"的边界
- 进度反馈用 `flush=True`，别让用户对着空白屏干等
- 单点失败不扩散：三层防护，每一层都把异常就地消化
- `if __name__ == "__main__":` 只做接线，逻辑放进类和函数才能被测试
