# 执行与撤销：让工具真正可信

> 生成计划只是纸上谈兵，真正移动文件才有价值——但真正动手才需要安全网。这一课讲清为什么 undo 是专业工具和脚本的分水岭。

## Executor：先写日志，再执行

第 26 课定下的原则：**先写日志，再执行**。日志是一份 JSON 文件，记录本次操作的每一条"源 → 目标"映射和时间戳。它是撤销的唯一依据，也是工具可信度的来源。

```python
import json
import shutil
from datetime import datetime
from pathlib import Path


class Executor:
    def __init__(self, root, log_dir):
        self.root = Path(root)
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(parents=True, exist_ok=True)

    def apply(self, plan):
        """plan: {dest_rel: src_abs}。执行前先写日志，再逐个移动。
        返回 (成功数, 失败列表)"""
        run_id = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
        log_file = self.log_dir / f"ops_{run_id}.json"

        # 1) 先把完整计划写成日志（核心：先落盘，再执行）
        log = {
            "run_id": run_id,
            "timestamp": datetime.now().isoformat(),
            "root": str(self.root),
            "operations": [
                {"src": src, "dest": str(self.root / dest), "status": "pending"}
                for dest, src in plan.items()
            ],
        }
        log_file.write_text(json.dumps(log, ensure_ascii=False, indent=2), encoding="utf-8")

        # 2) 逐个执行，实时更新日志状态
        ok, failed = 0, []
        for op in log["operations"]:
            try:
                d = Path(op["dest"])
                d.parent.mkdir(parents=True, exist_ok=True)
                shutil.move(op["src"], op["dest"])
                op["status"] = "done"
                ok += 1
            except OSError as e:
                op["status"] = "failed"
                op["error"] = str(e)
                failed.append(op["src"])

        # 3) 回写状态
        log_file.write_text(json.dumps(log, ensure_ascii=False, indent=2), encoding="utf-8")
        return ok, failed, run_id

    def undo(self, run_id=None):
        """撤销一次操作：读取日志，把每条 done 的移动反向执行。
        不传 run_id 时撤销最近一次。"""
        log_file = self._pick_log(run_id)
        log = json.loads(log_file.read_text(encoding="utf-8"))

        undone, failed = 0, []
        for op in reversed(log["operations"]):        # 反向执行，避免依赖已移走的父目录
            if op["status"] != "done":
                continue
            try:
                src = Path(op["dest"])
                dest = Path(op["src"])
                if src.exists():
                    dest.parent.mkdir(parents=True, exist_ok=True)
                    shutil.move(str(src), str(dest))
                    undone += 1
            except OSError as e:
                failed.append(op["dest"])
        return undone, failed

    def _pick_log(self, run_id):
        if run_id:
            return self.log_dir / f"ops_{run_id}.json"
        # 取最新的日志
        logs = sorted(self.log_dir.glob("ops_*.json"))
        if not logs:
            raise FileNotFoundError("没有可撤销的操作记录")
        return logs[-1]
```

**为什么先写日志**：即使执行到一半崩溃，日志里每一条都带着 `status`。重启后照样能判断"哪些移走了、哪些没动"，照着日志反向走就行。

**为什么 undo 要反向遍历**：先撤销最后移走的文件，它的父目录可能正是前面某次移动的目标，反向走能避开"父目录已被移走"的尴尬。

## 实战演示：预览 → 执行 → 撤销

```python
# 模拟计划（真实场景来自第 28 课的 plan_all）
plan = {
    "Docs/2026-09/a.pdf": "/d/a.pdf",
    "Images/2026-09/b.jpg": "/d/b.jpg",
}

ex = Executor("/d", "/d/.organize_logs")

# 预览阶段（不写文件，只打印）
print(f"[预览] 将移动 {len(plan)} 个文件")
for dest, src in plan.items():
    print(f"  {src}  ->  {dest}")

# 执行
ok, failed, run_id = ex.apply(plan)
print(f"✓ 完成 {ok} 个，失败 {len(failed)} 个，run_id={run_id}")

# 撤销
undone, errs = ex.undo(run_id)
print(f"✓ 已撤销 {undone} 个文件")
```

日志文件长这样：

## 翻车现场：直接覆盖同名文件，无法撤销

不检查目标是否存在，`shutil.move` 在目标已有时会**静默覆盖**（Unix 行为），文件就此消失。

```python
import shutil

# WRONG：目标存在时直接覆盖
def apply_wrong(plan):
    for dest, src in plan.items():
        shutil.move(src, dest)   # dest 已存在 → 源被覆盖丢失

plan = {"Docs/a.pdf": "/d/a.pdf"}
# /d/Docs/a.pdf 原本有内容 → 被 /d/a.pdf 覆盖，原内容永久丢失
```

```python
import shutil
from pathlib import Path

# 正确：执行前检查目标存在性，冲突按策略处理
def move_safe(src, dest, conflict="suffix"):
    d = Path(dest)
    if d.exists():
        if conflict == "skip":
            return False, "skip"
        if conflict == "suffix":
            stem, suffix = d.stem, d.suffix
            i = 1
            while d.exists():
                d = d.with_name(f"{stem}_{i}{suffix}")
                i += 1
        # overwrite 直接走 shutil.move，由调用方显式确认
    d.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(src, str(d))
    return True, str(d)

print(move_safe("/d/a.pdf", "/d/Docs/a.pdf"))     # (True, '/d/Docs/a.pdf')
print(move_safe("/d/b.pdf", "/d/Docs/a.pdf"))     # (True, '/d/Docs/a_1.pdf')
```

冲突策略在配置里定，执行层只是照办。这也说明**为什么 Planner 要预先生成唯一目标名**——把冲突解决放在执行之前，执行时才能纯粹地"搬"。

## 翻车现场：撤销时只读第一条日志，误撤销整个批次

日志目录里可能有多份记录。撤销时如果取错文件，要么撤销了别的操作，要么找不到目标。

```python
from pathlib import Path

# WRONG：硬编码一个日志文件名
def undo_wrong(log_dir):
    log = json.loads((Path(log_dir) / "ops.json").read_text())
    # 如果目录里有 ops_1.json、ops_2.json，永远撤销的是 ops.json —— 不存在或过期
    ...

# 正确：明确 run_id，缺省取最新
def pick_log(log_dir, run_id=None):
    d = Path(log_dir)
    if run_id:
        target = d / f"ops_{run_id}.json"
        if not target.exists():
            raise FileNotFoundError(f"找不到 run_id={run_id} 的操作记录")
        return target
    logs = sorted(d.glob("ops_*.json"))
    if not logs:
        raise FileNotFoundError("没有可撤销的操作记录")
    return logs[-1]   # 最新一份

print(pick_log("/d/.organize_logs"))
# /d/.organize_logs/ops_20260928_150000_123456.json
```

**专业工具的细节就在这里**：run_id 明确、找不到时给清晰报错，而不是默默做错事。

## 小结

- 执行顺序：**先写日志 → 再逐个移动 → 实时更新日志状态**
- 日志是 undo 的唯一依据，崩溃后也能恢复判断
- 撤销时**反向遍历**操作列表，避免父目录已被移走的依赖问题
- 执行前必须检查目标存在性，冲突策略由 Planner 预先解决
- run_id 明确、找不到日志时给清晰报错，是专业工具的分水岭
